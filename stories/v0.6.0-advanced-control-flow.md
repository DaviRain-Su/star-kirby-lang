## 技术设计

### Token 新增

```zig
pub const TokenType = enum {
    // ... existing ...
    FOR,      // for
    IN,       // in
    BREAK,    // break
    CONTINUE, // continue
};
```

### AST 新增

```zig
pub const ForStatement = struct {
    token: token_mod.Token,     // 'for' token
    variable: Identifier,        // 循环变量
    iterable: *Expression,       // 可迭代对象
    body: *BlockStatement,
};

pub const BreakStatement = struct {
    token: token_mod.Token,
};

pub const ContinueStatement = struct {
    token: token_mod.Token,
};
```

### 循环控制对象

```zig
pub const LoopControl = enum {
    break_signal,
    continue_signal,
};

pub const LoopControlObj = struct {
    control: LoopControl,
};
```

## 测试用例

```monkey
// for 循环
let arr = [1, 2, 3, 4, 5];
let sum = 0;
for (x in arr) {
    let sum = sum + x;
};
sum  // 15

// range 函数
for (i in range(5)) {
    puts(i);
}  // 0 1 2 3 4

// break
let i = 0;
while (true) {
    if (i >= 5) {
        break;
    };
    let i = i + 1;
};
i  // 5

// continue
let sum = 0;
for (i in range(10)) {
    if (i % 2 == 0) {
        continue;
    };
    let sum = sum + i;
};
sum  // 25 (1+3+5+7+9)

// map
let doubled = map([1, 2, 3], fn(x) { x * 2 });
// [2, 4, 6]

// filter
let evens = filter([1, 2, 3, 4, 5], fn(x) { x % 2 == 0 });
// [2, 4]

// reduce
let sum = reduce([1, 2, 3, 4, 5], fn(acc, x) { acc + x }, 0);
// 15
```
