## 技术设计

### 数学函数实现

```zig
fn builtinAbs(allocator: Allocator, args: []Object) !Object {
    if (args.len != 1) return makeError(allocator, "wrong number of arguments");
    if (args[0].objectType() != .integer) return makeError(allocator, "argument must be INTEGER");
    const value = args[0].integer.value;
    return Object{ .integer = .{ .value = if (value < 0) -value else value } };
}
```

### 交互式 REPL

```zig
fn runInteractiveRepl(allocator: Allocator, repl: *REPL) !void {
    while (true) {
        std.debug.print(">> ", .{});
        // Read line from stdin
        const line = readLine();
        if (std.mem.eql(u8, line, "exit")) break;
        const result = try repl.eval(line);
        std.debug.print("{s}\n", .{result});
    }
}
```

## 测试用例

```monkey
// 数学函数
abs(-5);           // 5
min(3, 7);         // 3
max(3, 7);         // 7
min([5, 2, 8, 1]); // 1
max([5, 2, 8, 1]); // 8
pow(2, 10);        // 1024
sqrt(16);          // 4
sum([1, 2, 3, 4, 5]); // 15

// 数组操作
reverse([1, 2, 3]);              // [3, 2, 1]
sort([3, 1, 4, 1, 5]);           // [1, 1, 3, 4, 5]
find([1, 2, 3, 4], fn(x) { x > 2 }); // 3
some([1, 2, 3], fn(x) { x > 2 }); // true
every([1, 2, 3], fn(x) { x > 0 }); // true
slice([1, 2, 3, 4, 5], 1, 4);    // [2, 3, 4]
concat([1, 2], [3, 4]);          // [1, 2, 3, 4]
flatten([[1, 2], [3, 4]]);       // [1, 2, 3, 4]

// 系统交互
getenv("HOME");    // "/home/user"
time();            // 1704268800000

// 类型转换
bool(0);           // false
bool(1);           // true
bool("");          // false
bool("hello");     // true
array("abc");      // ["a", "b", "c"]
```
